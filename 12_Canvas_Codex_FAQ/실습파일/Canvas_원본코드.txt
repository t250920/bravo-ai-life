    <script>
        // 아코디언 토글 함수 (키보드 Tab, Space/Enter 접근성 완벽 지원)
        function toggleAccordion(id, btn) {
            const content = document.getElementById(id);
            const isExpanded = btn.getAttribute('aria-expanded') === 'true';
            const icon = btn.querySelector('.icon-arrow');

            if (isExpanded) {
                btn.setAttribute('aria-expanded', 'false');
                content.classList.add('hidden');
                btn.classList.remove('bg-blue-50');
                if (icon) icon.classList.remove('rotate-180');
            } else {
                btn.setAttribute('aria-expanded', 'true');
                content.classList.remove('hidden');
                btn.classList.add('bg-blue-50');
                if (icon) icon.classList.add('rotate-180');
            }
        }

        // 시뮬레이터 로직
        document.addEventListener('DOMContentLoaded', () => {
            const dateSelect = document.getElementById('paymentDate');
            const timeSelect = document.getElementById('paymentTime');
            const statusSelect = document.getElementById('orderStatus');

            function updateSimulation() {
                const dateText = dateSelect.options[dateSelect.selectedIndex].text;
                const timeText = timeSelect.options[timeSelect.selectedIndex].text;
                const statusText = statusSelect.options[statusSelect.selectedIndex].text;

                const dateVal = dateSelect.value;
                const timeVal = timeSelect.value;
                const statusVal = statusSelect.value;

                // 화면에 선택된 정보 텍스트 표시
                document.getElementById('disp-info').textContent = `${dateText} / ${timeText} / ${statusText}`;

                let shippingMsg = "";
                let changeMsg = "";

                // 1. 발송 예정일 계산 로직
                if (dateVal === "10-30") {
                    // 10월 30일(금)은 v1.0 적용 (14:00 컷오프)
                    if (timeVal === "1") {
                        shippingMsg = "10월 30일(금) 당일 발송 예정";
                    } else {
                        // 14:00 이상은 다음 영업일(11월 2일 월요일)
                        shippingMsg = "11월 2일(월) 발송 예정 (다음 영업일)";
                    }
                } else if (dateVal === "11-02") {
                    // 11월 2일(월)은 v1.1 적용 (15:00 컷오프)
                    if (timeVal === "1" || timeVal === "2") {
                        // 15:00 미만이므로 당일 발송
                        shippingMsg = "11월 2일(월) 당일 발송 예정";
                    } else {
                        // 15:00 이상은 다음 영업일(11월 3일 화요일)
                        shippingMsg = "11월 3일(화) 발송 예정 (다음 영업일)";
                    }
                }

                // 2. 배송지 변경 가능 여부 로직
                if (statusVal === "unknown") {
                    changeMsg = "먼저 고객의 주문 상태를 확인해주세요.";
                } else if (statusVal === "before") {
                    changeMsg = "고객센터 접수 및 배송지 변경 가능";
                } else if (statusVal === "packing") {
                    changeMsg = "물류팀 확인 후 가능 여부 안내 (포장 중)";
                } else if (statusVal === "done") {
                    changeMsg = "고객센터 변경 불가 (발송 완료)";
                }

                // 결과 텍스트 적용
                document.getElementById('res-shipping').textContent = shippingMsg;
                document.getElementById('res-change').textContent = changeMsg;
            }

            // 이벤트 리스너 등록 및 초기 실행
            dateSelect.addEventListener('change', updateSimulation);
            timeSelect.addEventListener('change', updateSimulation);
            statusSelect.addEventListener('change', updateSimulation);
            
            updateSimulation(); // 페이지 로드 시 즉시 계산
        });
    </script>

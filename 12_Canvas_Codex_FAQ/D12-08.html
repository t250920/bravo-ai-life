③ 결과물 읽기

결과에는 적용 버전과 판단 근거를 함께 표시한다. 배송지 변경 판정에는 새 주소 대신 현재 주문 상태만 사용한다.

function evidence(date, status) {
const v = policy(date);
const shipping = v === "v1.0"
? "발송 v1.0: 14:00 미만"
: v === "v1.1"
? "발송 v1.1: 15:00 미만"
: "발송 기준: 미정";
return shipping + "\n배송지 변경 v1.0: " +
addressChange(status);
}
document.getElementById("evidence").textContent =
evidence("2026-11-02", "포장 중");
